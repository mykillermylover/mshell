```ts
import {
  type Body,
  type Contact,
  contactBetween,
  contactsWithFrame,
  type FrameSpec,
} from './liquid/geometry.ts';

/** Допуск закрытия выбранного контакта, px. */
const RESIDUAL = 0.5;

export interface SnapEnv {
  frame: FrameSpec;
  reach: number;
}

export interface SnapResult {
  x: number;
  y: number;
}

type ContactRef =
  | { kind: 'frame'; index: number }
  | { kind: 'body'; index: number };

interface RefContact {
  ref: ContactRef;
  contact: Contact;
}

interface Candidate {
  dx: number;
  dy: number;
  contacts: RefContact[];
  distance: number;
}

const moved = (body: Body, dx: number, dy: number): Body => ({
  ...body,
  x: body.x + dx,
  y: body.y + dy,
});

const contactFor = (
  ref: ContactRef,
  body: Body,
  env: SnapEnv,
  others: Body[],
): Contact | null => {
  if (ref.kind === 'frame') {
    return contactsWithFrame(body, env.frame)[ref.index] ?? null;
  }

  return contactBetween(body, others[ref.index]);
};

const allContacts = (
  self: Body,
  env: SnapEnv,
  others: Body[],
): RefContact[] => {
  const out: RefContact[] = [];

  for (const [index, contact] of contactsWithFrame(self, env.frame).entries()) {
    out.push({ ref: { kind: 'frame', index }, contact });
  }

  for (const [index, other] of others.entries()) {
    const contact = contactBetween(self, other);

    if (contact) {
      out.push({ ref: { kind: 'body', index }, contact });
    }
  }

  return out;
};

const make = (
  self: Body,
  env: SnapEnv,
  others: Body[],
  contacts: RefContact[],
): Candidate | null => {
  const first = contacts[0];

  if (!first) {
    return null;
  }

  const { contact } = first;
  const dx = contact.nx * contact.gap;
  const dy = contact.ny * contact.gap;
  const next = moved(self, dx, dy);

  for (const item of contacts) {
    const after = contactFor(item.ref, next, env, others);

    if (!after || Math.abs(after.gap) > RESIDUAL) {
      return null;
    }
  }

  return { dx, dy, contacts, distance: Math.hypot(dx, dy) };
};

const pick = (candidates: Candidate[]): Candidate | null => {
  let best: Candidate | null = null;

  for (const candidate of candidates) {
    if (!best || candidate.distance < best.distance) {
      best = candidate;
    }
  }

  return best;
};

export const snapTo = (
  env: SnapEnv,
  self: Body,
  others: Body[] = [],
): SnapResult | null => {
  const positives = allContacts(self, env, others).filter(
    ({ contact }) => contact.gap >= 0 && contact.gap <= env.reach,
  );

  if (positives.length === 0) {
    return null;
  }

  const candidates: Candidate[] = [];

  for (const item of positives) {
    const candidate = make(self, env, others, [item]);

    if (candidate) {
      candidates.push(candidate);
    }
  }

  const best = pick(candidates);

  return best ? { x: self.x + best.dx, y: self.y + best.dy } : null;
};
```

```ts
const make = (
  self: Body,
  env: SnapEnv,
  others: Body[],
  contacts: RefContact[],
): Candidate | null => {
  const first = contacts[0];

  if (!first) {
    return null;
  }

  let dx: number;
  let dy: number;

  if (contacts.length === 1) {
    dx = first.contact.nx * first.contact.gap;
    dy = first.contact.ny * first.contact.gap;
  } else {
    const [a, b] = contacts;
    const det =
      a.contact.nx * b.contact.ny - a.contact.ny * b.contact.nx;

    if (Math.abs(det) < 1e-6) {
      return null;

    dx =

    dy =

    if (!after || Math.abs(after.gap) > RESIDUAL) {
      return null;
    }
  }

  return {
    dx,
    dy,
    contacts,
    distance: Math.hypot(dx, dy),
    key: keyOf(contacts),
  };
```

```ts
const buildCandidates = (
  self: Body,
  env: SnapEnv,
  others: Body[],
  pool: RefContact[],

  const out: Candidate[] = [];

  for (const item of pool) {

    if (candidate) {
      out.push(candidate);
    }
  }

  for (let i = 0; i < pool.length; i++) {
    for (let j = i + 1; j < pool.length; j++) {
      const candidate = make(self, env, others, [pool[i], pool[j]]);
```

```ts
const eject = (
  self: Body,
  env: SnapEnv,
  others: Body[],
): SnapResult | null => {
  const candidates: Candidate[] = [];

  for (const other of others) {
    const overlapX =
      Math.min(self.x + self.w, other.x + other.w) -
      Math.max(self.x, other.x);
    const overlapY =
      Math.min(self.y + self.h, other.y + other.h) -
      Math.max(self.y, other.y);

    if (overlapX <= 0 || overlapY <= 0) {
      continue;
    }

    const pushes: Array<[number, number]> = [
      [-(overlapX + 0.5), 0],
      [overlapX + 0.5, 0],
      [0, -(overlapY + 0.5)],
      [0, overlapY + 0.5],
    ];

    for (const [dx, dy] of pushes) {
      const next = moved(self, dx, dy);

      if (!overlaps(next, env, others)) {
        candidates.push({
          dx,
          dy,
          contacts: [],
          distance: Math.hypot(dx, dy),
          key: '',
        });
      }
    }
  }

  const best = pick(candidates);

  return best ? { x: self.x + best.dx, y: self.y + best.dy } : null;
};
```

```ts
  constructor(
    readonly spec: PopSpec,
    private style: () => {
      border: number;
      screenRadius: number;
      revealDuration: number;
    },
    revealed: () => void,
  ) {
    const { revealDuration } = this.style();
    const revealer = new Gtk.Revealer({
      transitionType: SLIDE[spec.edge],
      transitionDuration: revealDuration,
      revealChild: false,
    });
    revealer.set_child(padded(spec.content()));
    revealer.connect('notify::child-revealed', revealed);
    this.revealer = revealer;
    this.apply();
  }

  /** Переставить слот и обновить длительность по текущему конфигу. */
  apply() {
    const { border, screenRadius, revealDuration } = this.style();
    this.revealer.transitionDuration = revealDuration;

    const along = this.spec.align ?? 'center';
    const offset = border + screenRadius;
    const horizontal =
      this.spec.edge === 'top' || this.spec.edge === 'bottom';
    const alignment = ALIGN[along];

    if (horizontal) {
      this.revealer.halign = alignment;
      this.revealer.valign =
        this.spec.edge === 'top' ? Gtk.Align.START : Gtk.Align.END;
      if (this.spec.edge === 'top') {
        this.revealer.marginTop = border;
      } else {
        this.revealer.marginBottom = border;
      }
      if (along === 'start') {
        this.revealer.marginStart = offset;
      }
      if (along === 'end') {
        this.revealer.marginEnd = offset;
      }
    } else {
      this.revealer.valign = alignment;
      this.revealer.halign =
        this.spec.edge === 'left' ? Gtk.Align.START : Gtk.Align.END;
      if (this.spec.edge === 'left') {
        this.revealer.marginStart = border;
      } else {
        this.revealer.marginEnd = border;
      }
      if (along === 'start') {
        this.revealer.marginTop = offset;
      }
      if (along === 'end') {
        this.revealer.marginBottom = offset;
      }
    }
  }
```
